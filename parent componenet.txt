public class AccountController {

    @AuraEnabled(cacheable = true)
    public static List<Account> accgetDetails(String searchKey,Integer pageSize,Integer offsetSize){

        return[SELECT Id,Name,Industry FROM Account
               WHERE Name LIKE : ('%' + searchKey + '%')
               ORDER BY Name
               LIMIT : pageSize
               OFFSET : offsetSize];
    }   
}

<template>
    <lightning-card title = "Pagination,RowAction,Searching">

        <lightning-input
            label = "Sarching"
            onchange = {searchHandle}>
        </lightning-input>

        <lightning-datatable
            key-field = "Id"
            data = {accounts}
            columns = {columns}
            onrowaction = {onRowAction}>
        </lightning-datatable>

        </lightning-button>

         <lightning-button

import { LightningElement, track } from 'lwc';
import {NavigationMixin} from 'lightning/navigation';
import accgetDetails from '@salesforce/apex/AccountController.accgetDetails';

const columns = [

    {label : 'Name' , fieldName : 'Name'},
    {label : 'Industry' , fieldName : 'Industry'},
    {
        type : 'action',
        typeAttributes : {
            rowActions : [
                {label : 'view' , name : 'view'},
                {label : 'edit' , name : 'edit'},
                {label: 'Show Contacts', name: 'showContacts'}
            ]
        }
    }
];
export default class ImperativePagination extends NavigationMixin(LightningElement) {

    columns = columns;
    @track accounts = [];

    searchKey = '';
    pageSize = 5;
    pageNumber = 1;
    selectedAccountId; 

    connectedCallback(){
        this.getAccountDetails();
    }

    getAccountDetails(){
        accgetDetails({
            searchKey : this.searchKey,
            pageSize : this.pageSize,
            offsetSize : (this.pageNumber-1)*this.pageSize
        }).then((res)=>{
            this.accounts = res;
        }).catch((err)=>{
            console.log('err'+err);
        })
    }

    searchHandle(event){
        this.searchKey = event.target.value;
        this.pageNumber = 1;
        this.getAccountDetails();
    }

    previousHandler(){
        if(this.pageNumber > 1){
            this.pageNumber--;
            this.getAccountDetails();
        }
    }

    nextHandler(){
        this.pageNumber++;
        this.getAccountDetails();
    }

    onRowAction(event){

        const row = event.detail.row;
    
        if(event.detail.action.name == 'view'){
            this[NavigationMixin.Navigate]({
                type : 'standard__recordPage',
                attributes :{
                    recordId : row.Id,
                    actionName : 'view'
                }
            })
        }
         if(event.detail.action.name == 'edit'){
            this[NavigationMixin.Navigate]({
                type : 'standard__recordPage',
                attributes :{
                    recordId : row.Id,
                    actionName : 'edit'
                }
            })
        }
        if(event.detail.action.name == 'showContacts'){
            this.selectedAccountId = row.Id; 
        }
    }
}
